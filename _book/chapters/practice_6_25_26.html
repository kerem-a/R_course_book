    • תחילה חשבו כל מדד בנפרד
    • לאחר מכן בעזרת פונקציית across
    • הוסיפו את הארגומנט .names לפונקציית across כדי להוסיף את המילה “_median” לשם של כל עמודה

  • חשבו את החציון עבור כל המשתנים הכמותיים בעזרת שימוש ב-where(is.numeric) שמרו את ה-DF בשם median_all_numeric

7. הרצת אחוזונים על הרבה משתנים עם reframe

  • בעזרת פקודת reframe ניצור df עם החישוב של חמישוני צריכת אנרגיה בחלוקה לפי קבוצות גיל. ננסה בהתחלה רק לשים את החישוב ואז נוסיף עוד עמודה עבור ההגדרה של החמישונים עצמם נעגל את התוצאה לספרה אחת אחרי הנקודה

    👈 reframe מאד מאד דומה ל-summarise אבל מתאימה למצבים בהם הפלט של החישוב שלנו הוא יותר מערך אחד (תחשבו על ההבדל בין התוצאה של ממוצע לעומת החישוב שעשינו עכשיו לכמה אחוזונים)

  • ניצור גרף עמודות

  • נשלב across לחישוב של האחוזונים של האנרגיה ונהפוך את החישוב לכזה עבור כל המקרו-נוטריינטים

  • נשנה את המבנה של הטבלה ל-long

  • נכין גרף עמודות לכל הנוטריינטים.

  • נוסיף קוד ששומר את הגרף בתקייה

  • נשנה שוב את המבנה של הטבלה כך שקבוצות הגיל הן העמודות שלנו

  • נכיר את הפקודה relocate להזזת מיקום העמודה macro_nut

  • נייצא כאקסל, נוסיף לשם של האקסל שלנו את התאריך של היום

#build a function to calculate the height from wight and bmi
fn_height_calc<- function(weight, bmi){
  height <- sqrt(weight/bmi)
  return(height)
}


#test the function
fn_height_calc(70, 25)
fn_height_calc(val_study_analysis_ready$weight[2], val_study_analysis_ready$bmi[2])

#apply the function to the data
val_study_analysis_ready <- val_study_analysis_ready %>% 
  mutate(height = fn_height_calc(weight, bmi))

#Build a function that change the default for na.rm to TRUE + round the result
fn_med_na_rm <- function(x){
 med<- median(x, na.rm = TRUE) #calculate the median with na.rm = TRUE
 return(round(med,1)) #round the result
}

#test the function
fn_med_na_rm(val_study_analysis_ready$food_energy)

#use the function on the data
median_macro_nuts2 <- val_study_analysis_ready %>% 
  group_by(age_group) %>% 
  summarise(
    across(.cols=all_of(macro_nuts),
           .fns=~fn_med_na_rm(.),
           .names = "{col}_median"
    )) %>% 
  ungroup() 


#let's build a function that convert yes no to logical
fn_y_n_to_logic<- function(x){
  return(if_else(x=="yes",TRUE,FALSE))
}
